class Node:
    def __init__(self, data):
        self.data = data
        self.next = None


class SinglyLinkedList:
    def __init__(self):
        self.head = None

    # Insert from start
    def insert_start(self, data):
        newnode = Node(data)
        newnode.next = self.head
        self.head = newnode

    # Delete from end
    def delete_end(self):
        if self.head is None:
            print("List is empty")
            return

        if self.head.next is None:
            self.head = None
            return

        temp = self.head
        while temp.next.next is not None:
            temp = temp.next

        temp.next = None

    # Sum of elements
    def sum_elements(self):
        total = 0
        temp = self.head

        while temp is not None:
            total = total + temp.data
            temp = temp.next

        print("Sum:", total)

    # Traverse
    def traverse(self):
        temp = self.head

        if self.head is None:
            print("List is empty")
            return

        print("Linked List:", end=" ")
        while temp is not None:
            print(temp.data, end=" -> ")
            temp = temp.next

        print("None")

    # Insert from end
    def insert_end(self, data):
        newnode = Node(data)

        if self.head is None:
            self.head = newnode
            return

        temp = self.head
        while temp.next is not None:
            temp = temp.next

        temp.next = newnode

    # Delete from position
    def delete_position(self, pos):
        if self.head is None:
            print("List is empty")
            return

        if pos < 1:
            print("Invalid position")
            return

        if pos == 1:
            self.head = self.head.next
            return

        temp = self.head

        for i in range(pos - 2):
            if temp.next is None:
                print("Invalid position")
                return
            temp = temp.next

        if temp.next is None:
            print("Invalid position")
            return

        temp.next = temp.next.next

    # Multiplication of all elements
    def multiply(self):
        product = 1
        temp = self.head

        while temp is not None:
            product = product * temp.data
            temp = temp.next

        print("Multiplication:", product)

    # Traverse
    def traverse(self):
        temp = self.head

        print("Linked List:", end=" ")
        while temp is not None:
            print(temp.data, end=" -> ")

        print("None")

class Node:
    def __init__(self, data):
        self.data = data
        self.next = None


class CircularLinkedList:
    def __init__(self):
        self.head = None

    # Insert from start
    def insert_start(self, data):
        newnode = Node(data)

        if self.head is None:
            self.head = newnode
            newnode.next = self.head
            return

        temp = self.head

        while temp.next != self.head:
            temp = temp.next

        newnode.next = self.head
        temp.next = newnode
        self.head = newnode

    # Insert from end
    def insert_end(self, data):
        newnode = Node(data)

        if self.head is None:
            self.head = newnode
            newnode.next = self.head
            return

        temp = self.head

        while temp.next != self.head:
            temp = temp.next

        temp.next = newnode
        newnode.next = self.head

    # Update specific position
    def update(self, pos, value):
        if self.head is None or pos < 1:
            print("Invalid position")
            return

        temp = self.head

        for i in range(pos - 1):
            temp = temp.next

            if temp == self.head:
                print("Invalid position")
                return

        temp.data = value

    # Traverse
    def traverse(self):
        if self.head is None:
            print("List is empty")
            return

        temp = self.head

        print("Circular Linked List:", end=" ")

        while True:
            print(temp.data, end=" -> ")
            temp = temp.next

            if temp == self.head:
                break

        print("(Back to Head)")
